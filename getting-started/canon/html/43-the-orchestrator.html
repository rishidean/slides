#!/usr/bin/env bash
set -euo pipefail
MAX=4; n=0
while grep -q '🔲' docs/Roadmap.md && (( n < MAX )); do
n=$((n+1)); echo "=== turn $n ==="
claude -p "$(cat prompt.md)" || { echo "session error"; break; }
# permissions live in .claude/settings.json — no flags needed
grep -q '⛔' docs/Roadmap.md && { echo "blocked"; break; }
done
echo "done — $n turn(s)"
